fn first_word(some_string: &String) -> usize {
    let bytes = some_string.as_bytes();
    
    for (i, &item) in bytes.iter().enumerate() {
        if item == b' ' {
            return i;
        }
    }
    
    return some_string.len();
}

fn main() {
    let mut s = String::from("Hi there");
    let idx = first_word(&s);
    s.clear();
}
